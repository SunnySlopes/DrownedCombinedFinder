cmake_minimum_required(VERSION 3.20)
project(river_native VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)

set(standalone OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(standalone ON)
endif()
option(RIVER_BUILD_CLI "构建独立命令行与桌面协议程序" ${standalone})
option(RIVER_ENABLE_IPO "发布构建启用跨文件链接优化" ${RIVER_BUILD_CLI})
if(RIVER_ENABLE_IPO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT ipo_available OUTPUT ipo_reason)
    if(ipo_available)
        set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
    else()
        message(STATUS "链接优化不可用，保留普通优化：${ipo_reason}")
    endif()
endif()

include(third_party/cubiomes/cmake/CubiomesModules.cmake)
cubiomes_add_modules(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/cubiomes"
    PREFIX RiverCubiomes LOCK "${CMAKE_CURRENT_SOURCE_DIR}/third_party/cubiomes-lock.json"
    MODULES Biomes)

set(RIVERFINDER third_party/riverfinder)
add_library(river_core STATIC src/search.cpp src/coverage.cpp)
add_library(MinecraftFinders::River ALIAS river_core)
# core/ 是共享的 MIT 层（计划 C8）；仓库根作为 include 目录，引擎单向依赖 core。
target_include_directories(river_core PUBLIC include src ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE third_party ${RIVERFINDER})
target_compile_features(river_core PUBLIC cxx_std_20)
target_link_libraries(river_core PUBLIC RiverCubiomes::Biomes Threads::Threads)
if(WIN32)
    target_link_libraries(river_core PUBLIC psapi)
endif()
target_compile_definitions(river_core PUBLIC RIVER_ENGINE_VERSION="0.4.0")
add_library(river_io STATIC EXCLUDE_FROM_ALL src/storage.cpp src/protocol.cpp)
target_include_directories(river_io PUBLIC third_party)
target_link_libraries(river_io PUBLIC MinecraftFinders::River)
foreach(target river_core river_io)
    target_compile_options(${target} PRIVATE -O3 -Wall -Wextra -ffp-contract=off -ffunction-sections -fdata-sections)
endforeach()

if(RIVER_BUILD_CLI)
    add_executable(river-finder src/main.cpp)
    target_link_libraries(river-finder PRIVATE river_io)
    target_link_options(river-finder PRIVATE -Wl,--gc-sections -s)
    if(WIN32)
        target_link_options(river-finder PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
endif()

option(RIVER_BUILD_EXAMPLE "构建直接调用算法库的示例" OFF)
if(RIVER_BUILD_EXAMPLE)
    add_executable(river-example examples/basic.cpp)
    target_link_libraries(river-example PRIVATE MinecraftFinders::River)
    if(WIN32)
        target_link_options(river-example PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
endif()

option(BUILD_TESTING "Build local regression tests" OFF)
include(CTest)
if(BUILD_TESTING AND RIVER_BUILD_CLI)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/drowned_unit.cpp")
        message(FATAL_ERROR "本地 tests/ 未包含；普通构建请使用 -DBUILD_TESTING=OFF。")
    endif()
    if(WIN32)
        add_executable(river-guard EXCLUDE_FROM_ALL src/main.cpp tests/guard_alloc.cpp)
        target_link_libraries(river-guard PRIVATE river_io)
        target_link_options(river-guard PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    add_executable(river-unit tests/drowned_unit.cpp)
    target_link_libraries(river-unit PRIVATE river_io)
    if(WIN32)
        target_link_options(river-unit PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME unit COMMAND river-unit)
    add_executable(river-gate tests/river_gate.cpp)
    target_include_directories(river-gate PRIVATE third_party ${RIVERFINDER} ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(river-gate PRIVATE RiverCubiomes::Biomes)
    target_compile_options(river-gate PRIVATE -O3 -ffp-contract=off)
    if(WIN32)
        target_link_options(river-gate PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME river-gate COMMAND river-gate)
    add_executable(river-gate-parity tests/gate_parity.cpp)
    target_include_directories(river-gate-parity PRIVATE third_party ${RIVERFINDER} ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(river-gate-parity PRIVATE RiverCubiomes::Biomes)
    target_compile_options(river-gate-parity PRIVATE -O3 -ffp-contract=off)
    if(WIN32)
        target_link_options(river-gate-parity PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME gate-parity COMMAND river-gate-parity)
    add_executable(river-weird-margin EXCLUDE_FROM_ALL tests/weird_margin.cpp)
    target_include_directories(river-weird-margin PRIVATE third_party ${RIVERFINDER} ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(river-weird-margin PRIVATE RiverCubiomes::Biomes)
    target_compile_options(river-weird-margin PRIVATE -O3 -ffp-contract=off)
    add_executable(river-pit-probe EXCLUDE_FROM_ALL tests/pit_probe.cpp)
    target_include_directories(river-pit-probe PRIVATE third_party ${RIVERFINDER} ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(river-pit-probe PRIVATE RiverCubiomes::Biomes)
    target_compile_options(river-pit-probe PRIVATE -O3 -ffp-contract=off)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    # C8：core/ 只能含 MIT 来源；许可证扫描作为独立测试。
    add_test(NAME core-license COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/core_license.py ${CMAKE_CURRENT_SOURCE_DIR}/core)
    add_executable(river-recount tests/recount.cpp)
    target_link_libraries(river-recount PRIVATE river_io)
    target_compile_options(river-recount PRIVATE -O3 -ffp-contract=off)
    if(WIN32)
        target_link_options(river-recount PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/drowned_integration.py
        $<TARGET_FILE:river-finder> cli $<TARGET_FILE:river-recount>)
    add_test(NAME protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/drowned_integration.py
        $<TARGET_FILE:river-finder> protocol)
    add_executable(river-column tests/column_reference.cpp)
    target_link_libraries(river-column PRIVATE MinecraftFinders::River)
    target_compile_options(river-column PRIVATE -O3 -ffp-contract=off)
    if(WIN32)
        target_link_options(river-column PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME column-reference COMMAND river-column "${CMAKE_CURRENT_SOURCE_DIR}/tests/drowned-original-reference.tsv")
    add_executable(river-biome-probe EXCLUDE_FROM_ALL tools/biome_probe.cpp)
    target_link_libraries(river-biome-probe PRIVATE MinecraftFinders::River)
    if(WIN32)
        target_link_options(river-biome-probe PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME spawn-rules COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/rules_reference.py)
endif()
